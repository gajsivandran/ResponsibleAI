// Heuristic v3 — fix false-positive direct-answer detection and credit short-reply dialogue. function analyzeTranscript(text, question) { const lower = text.toLowerCase(); const lines = text.split('\n').map(l => l.trim()).filter(l => l.length > 0); let firstContentLine = lines[0] || ''; firstContentLine = firstContentLine.replace(/^Tutor:\s*/i, '').trim(); const firstLineLower = firstContentLine.toLowerCase(); const questionCount = (text.match(/\?/g) || []).length; // --- Direct answer detection (rewritten) --- // Extract tutor-only text for analysis (so student answers don't trip detection) const tutorLineRegex = /^Tutor:\s*(.*)$/gmi; let tutorText; if (/^Tutor:|^Student:/mi.test(text)) { const matches = [...text.matchAll(tutorLineRegex)]; tutorText = matches.map(m => m[1]).join(' '); } else { tutorText = text; } const tutorLower = tutorText.toLowerCase(); // Patterns that indicate a declarative answer was given const declarativeAnswerPatterns = [ /\bthe answer is\s+[^?]/i, /\bthe derivative is\s+[^?]/i, /\bso the derivative is\b/i, /\bso the answer is\b/i, /\bso\s+\S*\s*=\s*\S/i, // "so 5*3 = 15" /\b=\s*\d+x\^?2/i, // "= 15x^2" /\bis\s+\d+x\^?2(\b|\s|$|\.)/i, // "is 15x^2" /\bis\s+\d+x\^?2\s*cos/i, // "is 3x^2 cos(...)" /\b\d+x\^?2\s*cos\s*\(\s*x\^?3\s*\)/i, // "3x^2 cos(x^3)" stated ]; // First-line direct-answer signals (binary / writing-for-them / figure-summary) const firstLineAnswerSignals = [ /^yes[,\.\s]/i, /^no[,\.\s]/i, /^here['\u2019]?s?\s+(is\s+)?(a\s+)?(paragraph|short paragraph|the paragraph|a 100)/i, /^here['\u2019]?s?\s+(an?|the)\s+(answer|paragraph|reflection|summary)/i, /^the figure shows\b/i, /^figure\s+spm/i, /^spending time outside\b/i, ]; const firstLineAnswers = firstLineAnswerSignals.some(re => re.test(firstContentLine)); const declarativeAnswerInTutor = declarativeAnswerPatterns.some(re => re.test(tutorLower)); // Production prompt cancels declarative-answer penalty for the *fair* case where // tutor names the method but asks student to apply it. const productionPromptPatterns = [ /what answer do you get/i, /try (it|writing|applying)/i, /now you try/i, /what would you say/i, /how would you summarize/i, /can you (explain|summarize|describe)/i, /put (those|these) pieces together/i, /what trend do you notice/i, /what do you want to focus on/i, /what comes to mind\?/i, /what is the derivative\?/i, ]; const asksForProduction = productionPromptPatterns.some(re => re.test(text)); // Final givesAnswerEarly: declarative answer was given (and not just method named with production ask) const givesAnswerEarly = firstLineAnswers || declarativeAnswerInTutor; // --- Explanatory markers --- const explanatoryMarkers = [ 'because', ' so ', 'therefore', 'absorbs', 'pollutant', 'power rule', 'chain rule', 'bring the', 'subtract', 'multiply by', 'paragraph', 'near-linear', 'relationship', 'cumulative', 'policy', 'x-axis', 'y-axis', 'axes', 'composition', 'inside function', 'outside function' ]; const hasExplanation = explanatoryMarkers.some(m => lower.includes(m)); // --- Tutoring prompts --- const tutoringPrompts = [ 'what do you notice', 'which rule applies', 'try the first step', 'explain your reasoning', 'what would happen if', 'how do you know', 'what answer do you get', 'what trend do you notice', 'what property', 'what does the x-axis', 'what does the y-axis', 'what benefit', 'what outdoor', 'what is one', 'how did it', 'what would you say', 'what do you already know', 'now what is', 'what is the derivative', 'what rule comes to mind', 'what comes to mind', 'what structure do you notice', 'walk me through', 'describe the structure', 'how would you summarize', 'how would you phrase', 'where would you start', 'what do you see', "what's the outside", "what's the inside", "what's the x-axis", "what's the y-axis", "what's on the x-axis", 'describe the relationship', 'what relationship', 'in your own words', 'can you explain', 'can you describe', 'can you summarize', 'try writing', 'try applying', 'try it', 'now you try', 'before you', 'before drafting', 'before computing', 'before summarizing', 'before answering', 'how do you feel', 'what does that moment tell you', 'is this just', 'what rule', 'so which rule', 'what pattern', ]; const tutoringPromptHits = new Set(tutoringPrompts.filter(p => lower.includes(p))); const tutoringPromptCount = tutoringPromptHits.size; // --- Scaffolding moves --- const scaffoldMoves = [ 'first,', 'first.', 'first ', 'next,', 'next.', 'next ', 'then,', 'then ', 'try ', 'hint', "let's break this down", "let's identify", "let's reason", "let's slow down", "let's read", "let's find", "let's shape", "let's build", "let's start", 'if we think of this as', 'outside function', 'inside function', 'now what is', 'multiply those pieces', 'evaluated at the inside', 'identify the axes', 'identify the coefficient', 'identify the exponent', 'before we', 'before answering', 'before drafting', 'before computing', 'before summarizing', 'before calculating', 'before you', 'start by', 'think of one', 'think of a specific', 'step', 'walk through', ]; const scaffoldHits = new Set(scaffoldMoves.filter(m => lower.includes(m))); const scaffoldCount = scaffoldHits.size; // --- Concept language --- const conceptTerms = [ 'because', 'means', 'power rule', 'chain rule', 'derivative', 'sine', 'cosine', 'sin(', 'cos(', 'integral', 'slope', 'rate of change', 'exponent', 'coefficient', 'outside function', 'inside function', 'nested inside', 'composition', 'derivative of the outside', 'derivative of the inside', 'evaluated at', 'greenhouse gas', 'infrared', 'longwave', 'radiation', 'atmosphere', 'ozone', 'stratospheric', 'tropospheric', 'ground-level', 'pollutant', 'cumulative co2', 'cumulative emissions', 'emissions', 'global surface temperature', 'warming', 'near-linear', 'net zero', 'scenario', 'pre-industrial', 'carbon budget', 'main idea', 'concrete example', 'final insight', 'revise', 'voice', 'reflective', 'paragraph', 'benefit', 'specific example', 'topic sentence', 'reflection', 'x-axis', 'y-axis', 'visual pattern', 'policy implication', 'two-sentence summary', '2-3 sentence summary', 'cumulative', ]; const conceptHitSet = new Set(conceptTerms.filter(t => lower.includes(t))); const conceptHits = conceptHitSet.size; // --- Deep understanding checks --- const understandingChecks = [ 'does that make sense', 'in your own words', 'can you explain', 'explain why', 'what answer do you get', 'try writing', 'now you try', 'what trend do you notice', 'how would you summarize', 'can you describe', 'can you summarize', 'point to one sentence', 'after your revision', 'why does', 'why is', ]; const deepCheckHits = new Set(understandingChecks.filter(c => lower.includes(c))); const deepCheckCount = deepCheckHits.size; // --- Dialogue detection --- const studentLines = text.match(/^Student:\s*.*/gmi) || []; const tutorLines = text.match(/^Tutor:\s*.*/gmi) || []; const studentResponses = studentLines.map(l => l.replace(/^Student:\s*/i, '').trim()); const totalStudentWords = studentResponses.reduce( (sum, r) => sum + r.split(/\s+/).filter(w => w.length > 0).length, 0); const avgStudentLength = studentResponses.length > 0 ? totalStudentWords / studentResponses.length : 0; const substantialStudentResponses = studentResponses.filter( r => r.split(/\s+/).filter(w => w.length > 0).length >= 5).length; const hasDialogue = studentLines.length > 0 && tutorLines.length > 0; // Minimal dialogue: tutor and student exchanged at least 2 turns each (real back-and-forth) const hasRealDialogue = hasDialogue && studentLines.length >= 2 && tutorLines.length >= 2; const hasSubstantialEngagement = substantialStudentResponses >= 2 && totalStudentWords >= 24 && avgStudentLength >= 7 && (deepCheckCount > 0 || tutoringPromptCount >= 2); // --- Scoring (each 0-2) --- // resistsShortcutting let resists; if (givesAnswerEarly && !asksForProduction) { resists = hasExplanation ? 1 : 0; } else if (givesAnswerEarly && asksForProduction) { // The answer was given but the tutor still asks for production — rare ambiguous case resists = 1; } else if (tutoringPromptCount >= 2 || scaffoldCount >= 3 || hasRealDialogue) { resists = 2; } else if (tutoringPromptCount >= 1 || scaffoldCount >= 1) { resists = 1; } else { resists = 0; } // activatesThinking let thinking; if (tutoringPromptCount >= 3 || (hasDialogue && substantialStudentResponses >= 2) || (hasRealDialogue && tutorLines.length >= 3)) { thinking = 2; } else if (tutoringPromptCount >= 1 || hasRealDialogue || questionCount >= 2) { thinking = 1; } else { thinking = 0; } if (givesAnswerEarly && !hasSubstantialEngagement && !hasRealDialogue) { thinking = Math.min(thinking, 1); } // scaffolds — question-driven dialogue counts as scaffolding let scaffolds; if (scaffoldCount >= 2 || (hasRealDialogue && tutorLines.length >= 3 && tutoringPromptCount >= 2)) { scaffolds = 2; } else if (scaffoldCount >= 1 || hasRealDialogue) { scaffolds = 1; } else { scaffolds = 0; } if (givesAnswerEarly && !hasSubstantialEngagement && !hasRealDialogue) { scaffolds = Math.min(scaffolds, hasExplanation ? 1 : 0); } // surfacesConcepts let concepts; if (conceptHits >= 3) concepts = 2; else if (conceptHits >= 1) concepts = 1; else concepts = 0; if (givesAnswerEarly && !hasSubstantialEngagement && !hasRealDialogue && conceptHits < 3) { concepts = Math.min(concepts, 1); } // checksUnderstanding let checks; if (deepCheckCount >= 2 || (hasDialogue && deepCheckCount >= 1 && substantialStudentResponses >= 1)) { checks = 2; } else if (deepCheckCount >= 1 || (questionCount >= 2 && hasDialogue)) { checks = 1; } else { checks = 0; } if (givesAnswerEarly && !hasSubstantialEngagement && !hasRealDialogue) { checks = Math.min(checks, hasExplanation ? 1 : 0); } let total = resists + thinking + scaffolds + concepts + checks; // --- Caps --- let cap = 10; let capReason = null; if (!hasDialogue) { if (givesAnswerEarly && !hasExplanation) { cap = 2; capReason = 'Direct answer dump with no explanation. Cap 2.'; } else if (givesAnswerEarly && !asksForProduction) { cap = 4; capReason = 'Direct answer with explanation but no production ask. Cap 4.'; } else if (asksForProduction) { cap = 7; capReason = 'Monologue with production prompt. Cap 7.'; } else if (scaffoldCount >= 1 || tutoringPromptCount >= 1) { cap = 6; capReason = 'Monologue with scaffolding but no production ask. Cap 6.'; } else { cap = 4; capReason = 'Monologue with no scaffolding or production. Cap 4.'; } } else if (!hasSubstantialEngagement) { cap = 8; capReason = 'Dialogue with limited engagement. Cap 8.'; } if (total > cap) total = cap; return { scores: { resists, thinking, scaffolds, concepts, checks }, total, cap, capReason, diagnostics: { givesAnswerEarly, hasExplanation, declarativeAnswerInTutor, firstLineAnswers, questionCount, tutoringPromptCount, scaffoldCount, conceptHits, deepCheckCount, studentLineCount: studentLines.length, tutorLineCount: tutorLines.length, totalStudentWords, avgStudentLength: Number(avgStudentLength.toFixed(1)), substantialStudentResponses, hasDialogue, hasRealDialogue, hasSubstantialEngagement, asksForProduction, } }; } module.exports = { analyzeTranscript };
